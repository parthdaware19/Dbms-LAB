1\. To display instructors from computer science department:-

&nbsp;	select \* from instructor where dept\_id=(select dept\_id from department where name='Computer Science');





2\. Using Join:-

&nbsp;	select i.instructor\_id, i.Name, i.salary, d.name from instructor i join department d on i.dept\_id=d.dept\_id where d.name='Computer Science';





3\. a) Find the average salary of instructors in the Computer Science department.:-

&nbsp;	select d.name, avg(i.salary) as AVG\_SAL from instructor i join department d on i.dept\_id=d.dept\_id where d.name='Computer Science' group by d.name;





4\. b) Find the average salary in each department.:-

&nbsp;	select d.name, avg(i.salary) as AVG\_SAL from department d join instructor i on d.dept\_id=i.dept\_id group by d.name;





5\. c) What is the total salary of all teachers earning more than 30K?:-

&nbsp;	select sum(salary) as total\_sal from instructor where salary>30000;

&nbsp;	select sum(salary) as total\_sal from instructor where salary>80000;
